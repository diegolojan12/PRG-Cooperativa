export class ParametroGlobal {
  actualizar(valor) {
    const nuevo = String(valor).trim();
    if (!/^\d{1,5}$/.test(nuevo) ||
        Number(nuevo) > 10000)
      throw errorDeDominio(
        'Los puntos deben ser un número
         entero entre 0 y 10000.');
    // CA-06: decimales se rechazan en el campo
    this.valor = String(Number(nuevo));
    return this.valor;
  }
}
