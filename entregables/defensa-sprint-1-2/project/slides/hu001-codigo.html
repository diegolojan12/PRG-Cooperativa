validarPermisos(modulos, catalogo) {
  const existentes = new Set(
    catalogo.map((m) => m.id));
  if (modulos.some((m) => !existentes.has(m)))
    throw errorDeDominio('Permisos no válidos.');
  if (modulos.some(
      (m) => !this.disponibles.includes(m)))
    throw errorDeDominio('Ese módulo no está…');
  if (!this.obligatorios.every(
      (m) => modulos.includes(m)))
    throw errorDeDominio('El rol debe…');
  return [...new Set(modulos)];
}

backend/src/dominio/acceso.mjs:69 · backend/src/api.mjs:1451 · frontend/src/app/core/auth/guards.ts:14

La regla de permisos vive en la clase Rol del dominio: no acepta módulos inexistentes, ni módulos fuera de los disponibles del rol, y exige los obligatorios. La ruta solo orquesta: verifica que quien llama tenga el módulo Roles, valida con el dominio, guarda en RolPermiso y deja el cambio en la bitácora con el valor anterior y el nuevo.
